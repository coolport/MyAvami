
# === DOCKERIZE MYAVAMI PROJECT ===
# Project structure:
# myavami/
# ├── client/   → Vite + React frontend (port 5173)
# └── server/   → Node + Express backend (port 5555)
# etc etc



# === Step 1: Create client/Dockerfile ===
Create a file at client/Dockerfile with this content:
--------------------------------------------------
FROM node:20-alpine AS builder

WORKDIR /app

COPY ./package.json ./package-lock.json ./
RUN npm ci

COPY . .
RUN npm run build

FROM nginx:alpine

COPY --from=builder /app/dist /usr/share/nginx/html
COPY ./nginx.conf /etc/nginx/conf.d/default.conf

EXPOSE 5173

CMD ["nginx", "-g", "daemon off;"]
--------------------------------------------------


# === Step 2: Create client/nginx.conf ===
Create a file at client/nginx.conf with this content:
--------------------------------------------------
server {
  listen 5173;
  server_name localhost;

  root /usr/share/nginx/html;
  index index.html;

  location / {
    try_files $uri $uri/ /index.html;
  }
}
--------------------------------------------------


# === Step 3: Create server/Dockerfile ===
Create a file at server/Dockerfile with this content:
--------------------------------------------------
FROM node:20-alpine

WORKDIR /app

COPY ./package.json ./package-lock.json ./
RUN npm ci

COPY . .

EXPOSE 5555

CMD ["npm", "start"]
--------------------------------------------------


# === Step 4: Create docker-compose.yml at project root ===
Create a file at docker-compose.yml with this content:
--------------------------------------------------
version: "3.8"

services:
  backend:
    build:
      context: ./server
    ports:
      - "5555:5555"
    environment:
      - NODE_ENV=production

  frontend:
    build:
      context: ./client
    ports:
      - "5173:5173"
    depends_on:
      - backend
--------------------------------------------------


# === Step 5 (Optional but Recommended): Add .dockerignore ===
Inside both /client and /server folders, create a file named .dockerignore with this content:
--------------------------------------------------
node_modules
dist
--------------------------------------------------


# === Step 6: Build & Run the Dockerized Project ===
Run this command from the project root:
--------------------------------------------------
docker compose up --build
--------------------------------------------------


# === Notes ===
- Frontend served via Nginx at http://localhost:5173
- Backend API served at http://localhost:5555
- Ensure your frontend points API requests to http://backend:5555 inside Docker network
- Example Vite .env:
VITE_API_URL=http://localhost:5555

- do not do anything with git , handle version control. do not change branches, or edit main, etc. 

# === END ===
